                          39.2 容量规划

                          一句话

                          容量规划的成本问题只有一个：买多满——容量按峰值买是浪费、按均值买是事故，最优水位由「峰均比、超配单价、欠配损失」三个数算出来，而不是由「利用率越高越好」这条 KPI 决定。

                          先划边界：本节管“买多少”，不管“算多少”

                          容量话题在全书有五个主场，各管一段，不要在这里找重复答案：

                          你的问题 去哪一节
                          需要多少节点？峰值怎么取、余量怎么留、四类资源怎么算 57.2 容量模型
                          故障冗余要买多少？N+1 与跨 AZ 的容量代价 57.3 冗余系数
                          用资源用量而不是 QPS 估算、多久会用完 82.2 容量估算
                          峰值 QPS 与拐点怎么压出来 34.5 性能测试与负载模型、57.1 压测方法论
                          容量换算成钱、超配与欠配怎么比 本节
                          用哪种付费方式买（按需 / 预留 / Spot / 弹性） 39.4 弹性与预留实例

                          🔑 一句话分工：57.2 容量模型 算出「要几台」，本节算出「这几台值多少钱、买多满」。两者用同一组峰值与水位，但一个输出节点数，一个输出月成本与买法。

                          三个误区，先拆掉

                          误区 现实
                          容量规划就是算峰值 QPS 峰值只是输入之一。输出必须是「节点数 × 买法 × 月成本 × 复查日期」四件套，缺了钱和日期就只是算术练习
                          利用率越高越省钱 利用率越高，排队放大越陡：70% 水位排队放大 3.3 倍，90% 是 10 倍。省下的容量费是线性的，事故损失是阶跃的
                          上了弹性就不用算容量了 弹性有追赶延迟与冷启动，还有每小时的溢价单价；比扩容更快的突发，只能靠预留容量接住

                          把容量折成钱：同一负载的三种买法

                          前提只有一个：同一份 24 小时负载曲线，三种买法给出三种账单。单节点安全容量按 100 QPS（已折算 70% 水位）、固定节点 800 元/月、弹性单价 = 固定折算到小时的 1.6 倍。

                          package main

import (
	"fmt"
	"math"
)

// 负载曲线：24 小时，单位 QPS（示例数据，用于演示口径，不是任何真实业务）
var curves = map[string][]float64{
	"企业 ERP":  {900, 850, 820, 800, 810, 880, 1000, 1100, 1200, 1250, 1200, 1000, 950, 1050, 1200, 1250, 1200, 1000, 900, 860, 840, 830, 820, 900},
	"电商":     {300, 260, 240, 230, 250, 350, 600, 900, 1200, 1400, 1500, 1600, 1500, 1450, 1400, 1500, 1700, 2000, 2300, 2600, 2800, 2200, 1500, 800},
	"报表/批处理": {150, 150, 150, 150, 150, 150, 150, 150, 400, 800, 1200, 1600, 1600, 1400, 1000, 600, 250, 200, 150, 150, 150, 150, 150, 150},
}

const (
	nodeCapQPS   = 100.0 // 单节点安全容量（已按 70% 水位折算）
	fixedPrice   = 800.0 // 固定容量：元/节点·月
	elasticMul   = 1.6   // 弹性单价 = 固定折算到小时的 1.6 倍
	daysPerMonth = 30.0
	monthHours   = 730.0
)

func mean(xs []float64) float64 { s := 0.0; for _, x := range xs { s += x }; return s / float64(len(xs)) }

func main() {
	elasticHour := fixedPrice / monthHours * elasticMul // 元/节点·小时

	fmt.Println("=== 一、同一负载的三种买法 ===")
	for _, name := range []string{"企业 ERP", "电商", "报表/批处理"} {
		load := curves[name]
		avg, peak := mean(load), 0.0
		for _, q := range load { if q > peak { peak = q } }
		fmt.Printf("\n%s：均值 %.0f QPS，峰值 %.0f QPS，峰均比 %.2f\n", name, avg, peak, peak/avg)
		fmt.Printf("%-24s %8s %10s %10s %10s\n", "买法", "月成本", "平均利用率", "每峰值QPS", "比按峰值省")

		aNodes := math.Ceil(peak / nodeCapQPS)
		aCost := aNodes * fixedPrice

		bestCost := math.Inf(1)
		for n := 0.0; n <= aNodes; n++ {
			cap := n * nodeCapQPS
			el := 0.0
			for _, q := range load { if q > cap { el += (q - cap) / nodeCapQPS } }
			if c := n*fixedPrice + el*elasticHour*daysPerMonth; c < bestCost { bestCost = c }
		}
		cCost := 0.0
		for _, q := range load { cCost += q / nodeCapQPS * elasticHour * daysPerMonth }

		fmt.Printf("%-24s %8.0f %9.1f%% %10.1f %9s\n", "A 按峰值固定买", aCost, avg/(aNodes*nodeCapQPS)*100, aCost/peak, "-")
		fmt.Printf("%-24s %8.0f %9.1f%% %10.1f %8.0f%%\n", "B 基线固定+弹性补峰", bestCost, avg/(bestCost/fixedPrice*nodeCapQPS)*100, bestCost/peak, (1-bestCost/aCost)*100)
		fmt.Printf("%-24s %8.0f %9.1f%% %10.1f %8.0f%%\n", "C 全部按小时弹性", cCost, avg/(cCost/fixedPrice*nodeCapQPS)*100, cCost/peak, (1-cCost/aCost)*100)
	}

	fmt.Println("\n=== 二、固定买还是弹性买：每天用几小时是分界线 ===")
	breakEvenHours := fixedPrice / (elasticHour * daysPerMonth)
	fmt.Printf("固定节点 %.0f 元/月；弹性节点 %.2f 元/小时 → 盈亏平衡：每天 %.1f 小时\n\n", fixedPrice, elasticHour, breakEvenHours)
	fmt.Printf("%-10s %-14s %-14s %s\n", "每天用量", "固定月成本", "弹性月成本", "更省")
	for _, h := range []float64{2, 4, 8, 12, 15, 16, 24} {
		fx, el := fixedPrice, h*daysPerMonth*elasticHour
		pick := "弹性"
		if fx < el { pick = "固定" }
		fmt.Printf("%-10.0f %-14.0f %-14.0f %s\n", h, fx, el, pick)
	}

	fmt.Println("\n=== 三、欠配的盈亏平衡：水位推到 90% 得省多少才不亏 ===")
	load := curves["电商"]
	peak := 0.0
	for _, q := range load { if q > peak { peak = q } }
	aCost := math.Ceil(peak/nodeCapQPS) * fixedPrice
	// 把容量配到峰值的 90%：允许峰值时段超载 10%
	n90 := math.Ceil(peak * 0.9 / nodeCapQPS)
	saving := aCost - n90*fixedPrice
	overflow := 0.0
	for _, q := range load { if q > n90*nodeCapQPS { overflow += q - n90*nodeCapQPS } }
	breakEvenLoss := saving / (overflow * daysPerMonth)
	fmt.Printf("按峰值配：%v 节点 %.0f 元/月；只配到峰值的 90%%：%v 节点 %.0f 元/月\n", int(math.Ceil(peak/nodeCapQPS)), aCost, int(n90), n90*fixedPrice)
	fmt.Printf("省下 %.0f 元/月，代价是每天 %.0f QPS·h 超载 → 盈亏平衡损失 = %.3f 元/QPS·h\n", saving, overflow, breakEvenLoss)
	fmt.Printf("%-16s %-16s %-16s %s\n", "超载损失/元/QPS·h", "损失/月", "净收益", "该不该省")
	for _, loss := range []float64{0.01, breakEvenLoss, 0.5, 3} {
		l := overflow * loss * daysPerMonth
		pick := "该省"
		if l > saving { pick = "不该省" }
		if math.Abs(l-saving) < 1 { pick = "持平" }
		fmt.Printf("%-16.3f %-16.0f %+-16.0f %s\n", loss, l, saving-l, pick)
	}

	fmt.Println("\n=== 四、把容量成本换算成可比的单位 ===")
	perQPSMonth := aCost / peak
	perMillion := perQPSMonth / (monthHours * 3600 / 1e6)
	fmt.Printf("固定：%.1f 元/峰值 QPS·月；全部弹性折算：%.1f 元/峰值 QPS·月\n", perQPSMonth, elasticHour*monthHours/nodeCapQPS)
	fmt.Printf("换算成请求：%.2f 元 / 百万请求（%.4f 元/千次）\n", perMillion, perMillion/1000)
	fmt.Printf("一个月 730 小时，若按峰值买而平均只用到 45.5%%，每承载 1 QPS 实际付 %.1f 元\n", perQPSMonth/0.455)
}

                          逐条读这份输出：

                          1. 峰均比决定省钱空间。ERP 峰均比 1.27，负载几乎全天贴着峰值，「基线 + 弹性」只省 18%；报表峰均比 3.44，同样的方法省 59%。峰越尖，弹性越值钱；峰越平，弹性越像给稳定的容量多付一层溢价。
                          2. 全弹性不是最省。三种负载里「基线 + 弹性」都低于「全部按小时弹性」；ERP 甚至出现全弹性比按峰值固定买还贵 19%。原因写在单价里：弹性按小时买，单价是固定折算的 1.6 倍，而 ERP 几乎全天都要用——溢价乘上 24 小时，就是比包月更贵。
                          3. 按峰值买不是错，是贵。电商按峰值买平均利用率 45.5%，等于一半容量在空转；它买来的是「峰值时段也不超过 70% 水位」（排队放大 3.3 倍），真要「不排队」还得再乘 1.5–2 倍容量。所以只在「欠配损失 > 空转成本」时才该按峰值买。
                          4. 把容量折成单价的两种口径：每峰值 QPS·月 3.3–9.9 元（三种负载 × 三种买法的区间）、每百万请求 3.04 元。这两个数才能拿去和 39.1 成本模型与单位经济 的单位成本、和收入侧的每单毛利放在同一张表里比；单价本身的计费维度见 5.6 云成本结构。

                          🔑 容量规划的成本结论只有一句：先算清「超配的单价」和「欠配的单价」，再决定水位。算不出欠配的单价，就只能靠拍脑袋定余量，然后要么浪费一半预算，要么在峰值当天救火。

                          固定还是弹性：每天用 15.2 小时是分界线

                          推导只有一行：弹性节点的月成本 = 每天小时数 × 30 × 弹性时价，令它等于固定节点的月价：

                          盈亏平衡小时数 = 固定月价 ÷ (弹性时价 × 30)
             = 800 ÷ (1.75 × 30) = 15.2 小时/天
                          你买了什么 分界线怎么用
                          一个节点每天需要 > 15.2 小时 买固定/包月；弹性溢价会吃掉全部节省
                          一个节点每天只用 < 15.2 小时（定时批处理、CI、开发环境） 用弹性/按需；每天 8 小时约省 47%，6 小时约省 60%
                          峰值只出现在 大促、月末结算、开盘 基线买固定，峰值用弹性；覆盖不到的极端值用排队/降级兜

                          这个数会随两个参数变：弹性溢价（此处 1.6）和固定月价。前提是固定侧没有叠加承诺折扣——固定资源如果再买 1 年承诺（39.4 弹性与预留实例），固定单价更低，分界线会进一步左移（溢价 2.5 倍时约 9.7 小时/天）。所以谈弹性之前先拿到两个真实单价：弹性时价、固定资源打完成诺折扣后的月价。

                          ⚠️ 反过来的坑：停机不等于不付费。云盘、固定 IP、负载均衡在实例关停后通常仍按整月/按月计费，「下班关停省一半」只对纯按小时计费的资源成立；先确认哪些资源随实例停而停。

                          欠配的盈亏平衡：水位该推到多高

                          「利用率越高越省钱」这条 KPI 的问题在于：它只看分子（省下的容量费），不看分母（欠配损失）。这里把「水位」定义为容量 ÷ 峰值：容量只配到峰值的 90%，意味着峰值时段有约 10% 的请求接不住。账是这样：

                          数 值 怎么来的
                          省下的容量 1600 元/月（28 → 26 节点） （28 − 26）节点 × 800 元 = 1600
                          代价 每天 200 QPS·h 超载 峰值时段被削掉的那一截
                          盈亏平衡损失 0.267 元/QPS·h 1600 ÷ (200 × 30)

                          0.267 元/QPS·h 换算成请求是 0.000074 元/请求（0.267 ÷ 3600）。也就是说：只有每次被削掉的请求，期望损失低于万分之 0.74 元，把水位推到 90% 才划算。 交易、支付、登录这类链路，任何一个被拒的请求期望损失都远高于这个数，所以结论很直接：水位不是靠 KPI 压出来的，是靠「欠配单价」算出来的；算不出单价，就别往 90% 推。

                          💡 两种「水位」别混：余量表的水位是单节点利用率（相对标称 143 QPS），这里的水位是容量 ÷ 峰值。前者管「单机跑多满」，后者管「总量够不够」，两者都靠「欠配单价」定价。

                          ⚠️ 这里的「损失」不一定是收入：降级（关掉推荐、只留核心链路）的损失远小于拒单，所以先做降级能力，再谈提高水位。降级能力本质上是在给欠配买保险，且保费比多买容量便宜。另外模型把超载简化成「接不住」，现实中更常见的是排队变慢——损失体现为转化率与超时率，不是错误码，所以你要估的是「每慢 1 秒损失多少」，而不是「拒 1 个请求损失多少」。

                          峰均比：决定买法的唯一数字

                          峰均比 典型场景 推荐买法 别做
                          1.2–1.5 企业 ERP、内部系统 固定容量，按峰值 × 水位买 全弹性（溢价吃光收益，实测 ERP 反而贵 19%）
                          2–3 电商、社交 基线固定 + 弹性补峰 按均值配（峰值时段直接过载）
                          3–5 报表、批处理、定时任务 小基线 + 弹性 + 错峰 按峰值固定买（利用率只有 29%）
                          > 5 月末结算、秒杀、大促 预约/错峰/排队/降级 用容量解决（成本按峰均比线性放大）

                          峰均比超过 5 之后，继续加容量的边际收益迅速归零：峰值只持续几小时，买来的容量 90% 时间在空转。此时正确动作是削峰，把容量问题变成产品问题（至于「为省容量改架构值不值」，判据见 39.6 成本驱动的架构决策）。例外只有一个：合同里写死了「不可排队」的强 SLA（例如交易撮合、支付清算），那就只能买容量，且要把赔付金额算进容量预算：

                          削峰手段 做法 代价 什么时候用
                          排队 请求进队列，按序处理 延迟上升，用户可能流失 后台任务、异步流程
                          降级 关非核心功能，保主链路 功能损失，要提前设计开关 突发流量、故障兜底
                          错峰 预约、批处理挪到低峰、限时活动 产品要改，用户要教育 报表、对账、数据同步

                          余量（水位）的价格表

                          余量不是道德问题，是价格问题。撑 1000 QPS 的峰值，按不同单节点利用率买（单节点标称 143 QPS、800 元/月；节点数按 57.2 容量模型 的公式试算，本节只看钱）：

                          目标水位 安全承载/节点 需要节点 月成本 排队放大 相对 90% 水位多花
                          90% 129 QPS 8 6400 10 倍 —
                          80% 114 QPS 9 7200 5 倍 +800（+12.5%）
                          70% 100 QPS 10 8000 3.3 倍 +1600（+25%）

                          一句话：用 25% 的容量成本，把排队放大从 10 倍换到 3.3 倍。 这笔买卖在交易链路上几乎总是划算，在离线任务上几乎总是不划算——同一个数字，不同的 SLO，答案不同。排队放大的推导见 34.2 性能建模 (利特尔法则、排队论)，水位的取法见 57.2 容量模型。

                          容量账本：四个必须写下来的数

                          容量规划最常见的失败不是算错，是半年后没人知道当初为什么买 28 台。每次规划落一份账本，至少五行：

                          字段 示例 为什么必须写
                          峰值口径 历史峰值 × 2（保守取值，见 57.2 容量模型），或压测拐点 2800 QPS 不写口径，下一个人会用日均值再算一遍
                          水位 70%（余量 30%） 决定节点数的隐形参数
                          单价 固定 800 元/节点·月，弹性 1.75 元/节点·小时 单价变了，买法结论就变
                          复查日期 每季度 / 大促前 30 天 容量会过期，账本不会自己提醒
                          单位成本 8 元/峰值 QPS·月 → 3.04 元/百万请求 唯一能与业务量、收入对比的口径

                          🔑 容量账本的价值不在「记录」，在逼你把假设写出来。一个「28 台」没有信息量；「峰值 2800 QPS（实测量级）、70% 水位、800 元/台、增长系数 1.2、下季度复查」才是可被质疑、可被继承的决策。

                          反模式

                          反模式 后果
                          用利用率 KPI 压容量 容量推高到 90% 水位，排队放大 10 倍，一次抖动就雪崩
                          按峰值买所有环境 生产、预发、测试都按峰值，非生产环境常年空转
                          用弹性代替容量规划 比扩容更快的突发只能靠预留容量接，接不住就是事故
                          把容量当纯技术问题，不算钱 算不出超配/欠配单价，余量只能拍脑袋
                          容量模型本身没建/没维护（算漏四类资源、一次算完不管、不记假设、拿压测峰值当安全容量） 见 57.2 容量模型 的反模式与四类资源
                          以为「停机就不付费」，把云盘 / 固定 IP / 负载均衡也按小时折算 这些资源在实例关停后通常仍按整月计费，「下班关停省一半」的账直接算空，容量预算系统性低估

                          自检

                          Q1 企业 ERP 的峰均比只有 1.27，为什么「全部按小时弹性」反而比「按峰值固定买」贵 19%？

                          ▸ 答案：因为弹性省的是闲置时间，而峰均比 1.27 意味着几乎没有闲置时间——负载全天贴着峰值，弹性优势无处发挥；同时弹性单价是按固定价折算的 1.6 倍，溢价乘满 24 小时就超过了包月价。判据是 15.2 小时/天的分界线：每天要用超过 15.2 小时的资源，买固定一定更便宜。弹性只对「用得少、用得散」的容量划算。

                          Q2 老板要求「把利用率从 45% 提到 90%，省一半容量费」，你怎么用两个数字回绝？

                          ▸ 答案：① 欠配的期望损失：按本节口径，水位从 100% 压到 90%（省 1600 元/月）的盈亏平衡损失是 0.267 元/QPS·h ≈ 0.000074 元/请求，交易链路一次拒绝的损失远高于此，所以「省下的钱 < 欠配损失」；② 排队放大：90% 水位的等待放大是 10 倍，70% 是 3.3 倍，延迟恶化会直接体现在转化率与 SLO 违约上。正确做法是先把「欠配损失」量化出来，再谈水位——没有欠配单价的利用率目标，只是把成本转成了事故。

                          Q3 15.2 小时/天的分界线是怎么来的？如果弹性溢价从 1.6 倍涨到 2.5 倍，分界线往哪边移？

                          ▸ 答案：分界线来自「弹性月成本 = 固定月成本」：h × 30 × 弹性时价 = 固定月价，即 h = 800 ÷ (1.75 × 30) = 15.2。溢价涨到 2.5 倍时弹性时价升到约 2.74 元，分界线降到 800 ÷ (2.74 × 30) ≈ 9.7 小时/天——弹性变贵，就该更早改成买固定。反过来，若固定价降或弹性折扣加大，分界线右移，弹性覆盖面变大。

                          Q4 电商大促峰均比 10，为什么「多买容量」不是正确答案？

                          ▸ 答案：峰均比 10 意味着峰值只持续很短时间，按峰值买的容量 90% 时间在空转，成本按峰均比线性放大而收益是零；而且大促的峰值本身是「产品设计出来的」（整点开抢、限时折扣），可以改。正确动作是削峰：预约/排队（把瞬时并发摊平）、降级（保主链路）、错峰（把非实时任务挪走）、限流（保护下游）。峰均比 > 5 之后，容量规划的上限是产品设计，不是机器数量。

                          Q5 为什么容量账本里「8 元/峰值 QPS·月」比「28 台机器」更有用？

                          ▸ 答案：「28 台」只在当前单价、当前水位、当前峰值口径下成立，换个前提就失效；「8 元/峰值 QPS·月」是可换算、可对比的量：乘业务峰值就得到容量预算，除以 730×3600 得到每请求成本（约 3.05×10⁻⁶ 元），再乘 10⁶ 就是每百万请求约 3.04 元，能直接和 39.1 成本模型与单位经济 的单位成本、和收入侧毛利放在同一张表里比。容量决策要能被继承，靠的是单价与假设，不是台数。
