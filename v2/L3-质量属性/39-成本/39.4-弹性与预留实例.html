                      39.4 弹性与预留实例

                      一句话

                      弹性和预留是同一笔预算的两端：弹性为「不确定」付溢价，预留为「确定」换折扣——最优采购不是二选一，而是把负载按可预测性切成三层（基线买承诺、波动买按需、可中断买 Spot），承诺量由承诺期内的最低负载决定，不由平均负载或峰值决定。

                      🔑 一句话分工：5.6 告诉你「有哪些买法、大约省多少」，本节告诉你「买多少、什么时候买、买错了怎么退」。

                      四个误区，先拆掉

                      误区 现实
                      先买预留，再慢慢优化 承诺是把「当前浪费」锁进合同：先删闲置、再降规格，基线才真实（顺序见 5.6 云成本结构）
                      折扣越大越划算 折扣越大，要求的确定性越高。3 年省 55% 的前提是「三年都用满 45% 以上」，用不满就比按需还贵
                      预留 = 省 35% 只有被利用的那部分才省。买了不用的小时，是按折后价照付的纯浪费
                      Spot 不能上生产 无状态、可重试、有中断处理（两分钟通知 + 优雅退出）的负载完全可以用；不能用的是有状态、长事务、强一致的负载

                      核心公式：盈亏平衡使用率 = 折后价 ÷ 按需价

                      这是本节唯一必须背下来的式子。承诺折扣本质是「批发」：你保证用量，厂商给你折扣；保证的量用不完，差价就变成罚款。

                      购买方式 折后单价（示例） 省下比例 盈亏平衡使用率 含义
                      按需 1.20 元/小时 0% 100% 基准，随时可停
                      1 年预留 / 节省计划（无预付） 0.78 元/小时 35% 65% 一年里用不满 65% 的时间，就比按需贵
                      3 年预留（无预付） 0.54 元/小时 55% 45% 承诺越长，要求的确定性越高
                      3 年预留（全预付） 0.54 元/小时 + 预付 55% 51%（含 8% 资金成本） 预付的钱有机会成本，平衡点被抬高

                      （单价是示例量级，用于演示口径；折扣量级见 5.6 云成本结构，本表只做一件事：把折扣量级换算成「必须用满多少」。）

                      3 年全预付那行的算法值得单独说：单实例预付 0.54 × 3 × 8760 = 14,191 元，省下的按需费用要按月折现（8% 年化、36 个月年金现值系数 31.91）。折现后盈亏平衡使用率从 45% 抬到 51%——多出来的 6 个百分点，就是「这笔钱本可以干别的」的价格。

                      💡 这里假设「全预付与无预付同价、只差付款时点」，所以 51% 只反映资金成本。真实报价里全预付通常还能再拿到更低的时价，平衡点会低于 51%——但别用「全预付更便宜」去抵消「业务可能下滑」，两笔账要分开算。

                      🔑 所以「买不买预留」的判断顺序是：先估「这个资源在承诺期内会被用多少比例」，再和「折后价 ÷ 按需价」比。估不出使用比例，就先别买——没有基线预测的预留，是拿现金流赌业务不下滑。

                      三层负载：承诺、按需、Spot 各买哪段

                      真实的负载不是一个数，而是三层叠加：恒常的基线、可预测的波动、不可预测的突发。三层的确定性不同，买法就该不同。示例：基线 100 台（全月）、波动 20 台（30% 时间）、突发 15 台（5% 时间），月 730 小时。

                      package main

import (
	"fmt"
	"math"
)

// 价格是「示例量级」，用于演示口径，不是任何厂商的报价。
const (
	onDemandHour = 1.20 // 按需：元/实例·小时
	ri1yHour     = 0.78 // 1 年预留、无预付（省 35%）
	ri3yHour     = 0.54 // 3 年预留（无预付省 55%；全预付的资金成本另算）
	spotHour     = 0.36 // Spot（省 70%，可中断）
	monthHours   = 730.0
	years3Hours  = 3 * 8760.0
)

func main() {
	fmt.Println("=== 一、承诺折扣的盈亏平衡：一年至少用多少小时才不亏 ===")
	type plan struct {
		name    string
		price   float64
		upfront bool
		note    string
	}
	for _, p := range []plan{
		{"按需", onDemandHour, false, "基准"},
		{"1 年 RI / SP（无预付）", ri1yHour, false, "用不满这个比例就比按需还贵"},
		{"3 年 RI（无预付）", ri3yHour, false, "承诺越长，要求的确定性越高"},
	} {
		be := p.price / onDemandHour
		fmt.Printf("%-24s 折后 %.2f 元/h  省 %2.0f%%  盈亏平衡使用率 %3.0f%%  %s\n",
			p.name, p.price, (1-p.price/onDemandHour)*100, be*100, p.note)
	}

	// 3 年全预付的资金成本：预付一笔钱，换每小时折扣
	prepay := ri3yHour * years3Hours // 单实例 3 年全预付总额
	const annualRate = 0.08
	monthlyRate := annualRate / 12
	// 36 个月年金现值系数
	af := (1 - math.Pow(1+monthlyRate, -36)) / monthlyRate
	monthlyOnDemand := onDemandHour * monthHours // 全时按需的月成本
	beWithCash := prepay / (monthlyOnDemand * af)
	fmt.Printf("\n3 年全预付：单实例预付 %.0f 元；按 %.0f%% 年化资金成本折现，盈亏平衡使用率从 45%% 抬到 %.0f%%\n",
		prepay, annualRate*100, beWithCash*100)
	fmt.Println("→ 承诺期越长，越要先扣掉“这笔钱本可以干别的”")

	fmt.Println("\n=== 二、三层负载：哪些该买承诺，哪些该留弹性 ===")
	const (
		baseUnits  = 100.0 // 基线：恒常 100 台
		waveUnits  = 20.0  // 波动：20 台，30% 的时间需要
		burstUnits = 15.0  // 突发：15 台，5% 的时间需要
	)
	baseH, waveH, burstH := monthHours, 0.30*monthHours, 0.05*monthHours

	plans := []struct {
		name string
		cost float64
		note string
	}{
		{"A 全部按需", baseUnits*baseH*onDemandHour + waveUnits*waveH*onDemandHour + burstUnits*burstH*onDemandHour, "零承诺，成本最高"},
		{"B 按峰值全买 1 年 RI", (baseUnits+waveUnits+burstUnits)*baseH*ri1yHour, "峰值也承诺，空转严重"},
		{"C 基线 RI + 波动按需 + 突发 Spot", baseUnits*baseH*ri1yHour + waveUnits*waveH*onDemandHour + burstUnits*burstH*spotHour, "按“确定性”分层的标准组合"},
		{"D 基线 3 年 RI + 波动按需 + 突发 Spot", baseUnits*baseH*ri3yHour + waveUnits*waveH*onDemandHour + burstUnits*burstH*spotHour, "最省，但锁 3 年"},
	}
	fmt.Printf("%-38s %10s %12s %s\n", "方案", "月成本", "比按需省", "代价")
	base := plans[0].cost
	for _, p := range plans {
		fmt.Printf("%-38s %10.0f %11.0f%% %s\n", p.name, p.cost, (1-p.cost/base)*100, p.note)
	}

	fmt.Println("\n=== 三、承诺覆盖率：买到基线的百分之多少最省 ===")
	for _, ratio := range []float64{0.5, 0.7, 0.9, 1.0, 1.1, 1.2} {
		commit := baseUnits * ratio
		cost := commit*baseH*ri1yHour + waveUnits*waveH*onDemandHour + burstUnits*burstH*spotHour
		if commit < baseUnits { // 没承诺的基线用按需
			cost += (baseUnits - commit) * baseH * onDemandHour
		}
		waste := 0.0
		if commit > baseUnits {
			waste = (commit - baseUnits) * baseH * ri1yHour
		}
		note := "最优区间"
		if ratio < 0.9 {
			note = "承诺不足，基线部分按需付费"
		}
		if ratio > 1.0 {
			note = "超出基线的承诺必然空转"
		}
		fmt.Printf("承诺 %3.0f%%（%3.0f 台）  月成本 %6.0f 元  空转 %5.0f 元  %s\n", ratio*100, commit, cost, waste, note)
	}

	fmt.Println("\n=== 四、业务下坡时的承诺风险 ===")
	for _, drop := range []float64{0.0, 0.2, 0.4} {
		real := baseUnits * (1 - drop)
		committed := baseUnits // 按原基线买了 1 年
		waste := (committed - real) * baseH * ri1yHour
		if waste < 0 {
			waste = 0
		}
		fmt.Printf("基线下降 %2.0f%%：仍需为 %v 台的承诺付费，空转成本 %.0f 元/月，占承诺支出 %.0f%%\n",
			drop*100, int(committed), waste, waste/(committed*baseH*ri1yHour)*100)
	}
}

                      逐条读这份输出：

                      1. 按峰值买承诺是最典型的错误。B 方案把基线 100 + 波动 20 + 突发 15 全买了 1 年，省 18%；C 方案只承诺基线、波动按需、突发 Spot，省 33%——同一份负载，多省了 15 个百分点，靠的是「不承诺不确定的部分」。 承诺买的是确定性，把不确定的部分也承诺掉，等于用折扣价买空转。
                      2. 覆盖率有硬上限：100% 的基线。110% 起空转 5,694 元/月，120% 空转 11,388 元/月，而且这只是「稳态不下滑」的理想情况。
                      3. 资金成本会改变结论。3 年全预付看起来省 55%，把 8% 的资金成本折进去，盈亏平衡使用率从 45% 抬到 51%——预付越多、承诺越长，越要先扣掉机会成本。
                      4. 风险是单边的。承诺只保护厂商：基线下降 20%，空转 11,388 元/月，占承诺支出 20%；下降 40% 就是 40%。这正是「覆盖率上限 80%–90%」的来源——留下 10%–20% 不承诺，买的是业务下坡时的呼吸空间。

                      ⚠️ 模型里基线是恒定的，所以 100% 覆盖数学上最优；现实中基线的估计有误差、会衰减、会因架构调整而消失，所以实际操作取 80%–90%。这不是免费的保险：本例 90% 覆盖月成本 65,459 元，比 100% 覆盖的 62,393 元每月多付 3,066 元（约 5%）——这就是「买业务下坡时的呼吸空间」的期权费。模型给方向，留白有价格，按你对业务波动的判断买。

                      弹性不是免费的：四笔隐性成本

                      「用弹性替代预留」听起来是「只有用才付钱」，但弹性有自己的账单：

                      隐性成本 量级 怎么算进决策
                      弹性溢价 按秒/按小时单价通常高于包月折算（本节示例 1.6 倍） 用 39.2 容量规划 的 15.2 小时/天分界线：每天用超 15.2 小时就别用弹性
                      追赶延迟 实例 30 秒–5 分钟，冷启动更久（量级见 5.4 弹性伸缩与自动扩缩容） 比扩容更快的突发必须靠预留容量接，否则就是事故
                      伸缩抖动 实例频繁创建销毁（churn），带存储的实例启动更慢、更贵 设扩容快、缩容慢（冷却期），用 37.6 弹性伸缩 的策略
                      工程成本 预热、健康检查、指标口径、多可用区调度 这部分是人力，进 TCO，不是账单

                      🔑 弹性和预留的边界，用一句话划清：「这个资源每天会被用超过 15.2 小时吗？」 会 → 买承诺；不会 → 用弹性；不确定 → 先按需跑一个计费周期，拿真实数据再决定。

                      落地：三朵云的承诺折扣怎么买、怎么验、怎么退

                      指名与替代：AWS 用 EC2 Reserved Instances / Savings Plans，等效替代是 GCP 的 Committed Use Discounts、阿里云的节省计划 / 预留实例券。三家都在做同一件事：用承诺换折扣，差别在「承诺的是实例、是规格族、还是每小时花费」。

                      可抄（查承诺利用率，先看再买）：

                      # AWS：查本月预留实例利用率（Cost Explorer 需在账号内开启）
aws ce get-reservation-utilization \
  --time-period Start=2026-10-01,End=2026-11-01 \
  --granularity MONTHLY

# Savings Plans 是另一套接口，别只查 RI 就下结论
aws ce get-savings-plans-utilization \
  --time-period Start=2026-10-01,End=2026-11-01 \
  --granularity MONTHLY

                      可验：返回 JSON 里看 Total.UtilizationPercentage（SP 接口为 SavingsPlansUtilizationsByTime 下的同名指标）。低于 80% 说明买多了（承诺没用满），先降覆盖率再谈新增；接近 100% 且仍在按需付费，才是该补承诺的信号。两个接口都要查：只查 RI 会把 Savings Plans 的利用率漏掉。

                      可退：预留通常不可退，所以三条退路必须提前设计——① 优先买期限 1 年、无预付或部分预付，把最大损失限定在一年；② 用可转换 / 可灵活匹配的形态（Savings Plans 按花费承诺、可转换 RI），别绑死实例规格；③ 覆盖率上限 80%–90%，永远留一段按需额度当缓冲。

                      ⚠️ 未验证：本节环境没有云账号，本节命令与字段名按公开文档给出，执行前以当前 CLI 版本为准：ce 是全局端点（固定 us-east-1），与资源所在区域无关；Cost Explorer API 按调用次数计费，脚本轮询要把这笔钱算进去。本书真正跑通的是本节的成本模型。

                      到期与切换：承诺有到期日，也有「技术换代」风险——实例族升级、容器化改造、业务迁移都可能让当年买的规格作废。做一张承诺日历：到期前 90 天评估续不续、60 天定基线、30 天走审批。到期日与基线复查日绑在一起，才不会出现「技术已经换了两代，承诺还在为旧规格付钱」；这张日历怎么进月度治理节奏，见 39.5 FinOps 与账单治理。

                      反模式

                      反模式 后果
                      没算基线就买 3 年承诺 业务下坡或架构调整时，承诺变成沉没成本（示例：基线降 20% 就空转 11,388 元/月）
                      按峰值买承诺 峰值只占少数时间，承诺价照付，省下的 18% 远不如分层组合的 33%（同类：把预留买到峰值的 100%，见 5.6 云成本结构）
                      把 Spot 当按需用 中断处理没做，任务被回收后重跑，成本反而翻倍
                      只看折扣率，不看使用率 省 55% 的 3 年承诺，若只用到 40%，实际比按需还贵
                      承诺不设到期日历 到期自动转按需，账单悄悄跳升；或过期后忘记续，白白失去折扣
                      弹性开到最大档 缩容不设冷却期，实例反复创建销毁，churn 成本吃掉节省
                      • 相邻：57.4 FinOps 实践（三步闭环与三把刀）

                      自检

                      Q1 1 年预留省 35%、3 年预留省 55%，为什么 3 年的盈亏平衡使用率反而更低（45% vs 65%），却更容易买错？

                      ▸ 答案：盈亏平衡使用率 = 折后价 ÷ 按需价：1 年是 0.78/1.20 = 65%，3 年是 0.54/1.20 = 45%，所以 3 年「要求的用量下限」更低——这是折扣给的让步。但它锁定期更长：1 年买错只亏一年，3 年买错要亏三年，而且三年的时间里技术栈、实例族、业务形态都可能变。平衡点低 ≠ 风险低；期限越长，越要先扣资金成本（示例把 45% 抬到 51%），再算「业务能不能稳三年」。

                      Q2 同一条负载曲线，为什么「按峰值全买预留」只省 18%，而「基线承诺 + 波动按需 + 突发 Spot」能省 33%？

                      ▸ 答案：因为承诺的收益只来自被用满的小时。按峰值买 135 台，波动与突发那 35 台大部分时间空转，是按折后价照付的浪费；分层组合只承诺基线 100 台（几乎 100% 被用满），波动用多少付多少，突发只在大约 5% 的时间用 Spot。省钱的来源不是折扣率，是「承诺量与真实用量对齐」。

                      Q3 覆盖率买到基线的 120%，看起来「更保险」，为什么是错的？

                      ▸ 答案：承诺只对厂商有保护作用，对买方是单边义务：承诺 120 台，超出的 20 台全月空转，示例里每月浪费 11,388 元；而业务下坡 20% 时，同样的空转成本会原样出现。覆盖率的正确上限是「承诺期内最低负载」，工程上取基线的 80%–90%，留出的 10%–20% 是按需额度，也是业务收缩时的缓冲。

                      Q4 一个团队说「我们全用弹性，只有用的时候才付钱，所以最便宜」。哪两个数能反驳他？

                      ▸ 答案：① 弹性溢价：本节示例弹性时价是固定折算的 1.6 倍，用 39.2 容量规划 的 15.2 小时/天分界线——每天用超过 15.2 小时的资源，弹性一定更贵；② 追赶延迟：弹性扩容有追赶延迟，冷启动更久（量级见 5.4 弹性伸缩与自动扩缩容），突发超出这个速度时只能靠预留容量接，接不住就是事故。「只为用到的付费」不等于「总支出更低」——溢价和延迟都是账单的一部分。

                      Q5 为什么承诺折扣要设「到期日历」，而不是每年年底统一续？

                      ▸ 答案：因为承诺绑定的是当时的规格与架构：实例族升级、容器化、业务迁移都会让旧承诺作废，年底统一续等于让旧规格多付一年钱；同时到期日错过后会自动转按需，账单跳升且没人负责。到期前 90/60/30 天三段评估（续不续、基线多少、谁审批）把「续费」变成一次有数据的架构复核，而不是一次财务动作。
