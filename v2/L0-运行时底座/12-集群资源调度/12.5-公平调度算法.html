                  12.5 公平调度算法 (MMF / DRF)

                  一句话

                  公平不是 “平均分”，而是 “每个用户的主导航资源份额尽量相等”：MMF 按最稀缺的那一维分配，DRF 把多维资源压成 “主导份额（dominant share）”再比较——这样 “只要 1 核但吃 8G 内存” 与 “要 8 核但只吃 1G” 的任务才可比。

                  三种分配策略

                  策略 规则 问题
                  FIFO 先到先得，整块满足 大任务堵住后来者，短任务饿死
                  均分（Equal share） 每个用户 1/N 忽略 “任务形状不同”，形状不匹配时浪费严重
                  MMF（最大最小公平） 先满足最小需求，再逐步抬高 单资源维度公平，多维需扩展
                  DRF（主导资源公平） 按 max(cpu份额, mem份额) 排序，份额小者优先 需要知道每个用户的总份额

                  DRF 是 MMF 在多维资源上的推广：把 “用户 u 已占用的资源向量” 归一化成 “主导份额” d(u)=max(r_i/cap_i)，每次把资源给 d(u) 最小的用户。

                  实测：FIFO 与 DRF 在同一负载下的差别

                  双资源（16 核 / 16G），4 个用户形状不同的任务（A:1核4G、B:4核1G、C/D:1核1G），连续提交 400 个：

                  策略 成功服务任务数 主导份额（A/B/C/D）
                  FIFO 10 0.50 / 0.50 / 0.19 / 0.19
                  DRF 13 0.25 / 0.25 / 0.38 / 0.31

                  两个结论：① DRF 在同一资源池里多服务了 30% 的任务（13 vs 10）；② 它把 A、B 的份额从 0.5 压到 0.25，让 C、D 也有机会——这就是 “按主导份额排序” 的直接效果。

                  # DRF 选择：主导份额最小的用户优先
cand = [(max((used[u][0]+c)/CAP[0], (used[u][1]+m)/CAP[1]), u)
        for u, (c, m) in shapes.items() if fits(u, c, m)]
u = min(cand)[1]          # 份额最小者获得本次资源

                  DRF 算例：CPU/内存两种资源怎么分

                  先给一个小到能手算的集群：12 核 CPU、24 GiB 内存，两个资源维度；3 个用户各有无限 backlog，任务形状固定——A 是内存型 (1 核, 4 GiB)，B 是 CPU 型 (3 核, 1 GiB)，C 均衡 (2 核, 2 GiB)。同样“一个任务”，占掉的集群主导份额并不相同，这是整个算例的出发点：

                  用户 任务形状 CPU 份额 内存份额 主导资源 每任务 Δd
                  A (1 核, 4 GiB) 1/12 = 0.083 4/24 = 0.167 内存 0.167
                  B (3 核, 1 GiB) 3/12 = 0.250 1/24 = 0.042 CPU 0.250
                  C (2 核, 2 GiB) 2/12 = 0.167 2/24 = 0.083 CPU 0.167

                  规则只有三步：① 把每个用户的已用资源归一化成 d(u) = max(CPU 份额, 内存份额)；② 候选只在“剩余资源还放得下它下一个任务”的用户里挑；③ 选 d(u) 最小者分配，分完重算。没有任何候选时停止——这就是 max-min 公平点，而不是“份额正好相等”。第 ② 步不能省：只按 d(u) 选、不判放得下，会把任务分给一个装不进去的用户，调度器只能空转。

                  下面这段 Go 程序把三件事一次算完：DRF 的逐部分配序列、均分（固定配额、不借用）与 FIFO（按到达顺序、放不下就跳过）的对照。环境 go1.27.1 / darwin/arm64 / Apple M1，go run main.go：

                  // 12.5 公平调度算法 · DRF 算例
// 集群：12 核 CPU / 24 GiB 内存。3 个用户、2 维资源、无限 backlog。
// A(1,4) 内存型，B(3,1) CPU 型，C(2,2) 均衡型。
// 输出：单任务主导份额 → DRF 分配序列 → 终局份额 → DRF/均分/FIFO 三策略对比。
package main

import "fmt"

type res struct{ cpu, mem float64 }

type user struct {
	name string
	task res // 单个任务形状
	used res // 已分配累计
	n    int // 已服务任务数
}

var capRes = res{12, 24} // 集群容量（12 核 / 24 GiB）

func (r res) plus(o res) res  { return res{r.cpu + o.cpu, r.mem + o.mem} }
func (r res) fits(c res) bool { return r.cpu <= c.cpu && r.mem <= c.mem }

// share：主导份额 = max(cpu/总CPU, mem/总内存)
func (r res) share() float64 {
	s := r.cpu / capRes.cpu
	if m := r.mem / capRes.mem; m > s {
		s = m
	}
	return s
}

func newUsers() []*user {
	return []*user{
		{name: "A", task: res{1, 4}},
		{name: "B", task: res{3, 1}},
		{name: "C", task: res{2, 2}},
	}
}

func usedAll(us []*user) res {
	var t res
	for _, u := range us {
		t = t.plus(u.used)
	}
	return t
}

func served(us []*user) int {
	n := 0
	for _, u := range us {
		n += u.n
	}
	return n
}

func main() {
	us := newUsers()
	fmt.Println("== 形状 → 主导份额 ==")
	for _, u := range us {
		fmt.Printf("  %s 任务 (%g核,%gG)  cpu份额 %.3f  内存份额 %.3f  主导资源=%s  Δd=%.3f\n",
			u.name, u.task.cpu, u.task.mem, u.task.cpu/capRes.cpu, u.task.mem/capRes.mem,
			dominant(u.task.cpu/capRes.cpu >= u.task.mem/capRes.mem), u.task.share())
	}

	fmt.Println("\n== DRF 分配序列（每步选主导份额最小且放得下者）==")
	for step := 1; ; step++ {
		var pick *user
		for _, u := range us {
			if !usedAll(us).plus(u.task).fits(capRes) {
				continue // 放不下，不能成为候选
			}
			if pick == nil || u.used.share() < pick.used.share()-1e-12 {
				pick = u // 主导份额最小者优先
			}
		}
		if pick == nil {
			break // 没有任何人的下一个任务放得下 → max-min 公平点
		}
		before := pick.used.share()
		pick.used = pick.used.plus(pick.task)
		pick.n++
		fmt.Printf("  第 %d 步 选中 %s  d %.3f→%.3f  该用户累计 (%g核,%gG)\n",
			step, pick.name, before, pick.used.share(), pick.used.cpu, pick.used.mem)
	}

	t := usedAll(us)
	fmt.Printf("\n== DRF 终局 ==\n")
	for _, u := range us {
		fmt.Printf("  %s 任务 %d 个  (%g核,%gG)  d=%.3f\n", u.name, u.n, u.used.cpu, u.used.mem, u.used.share())
	}
	fmt.Printf("  合计 (%g核,%gG) 剩余 (%g核,%gG) → 三个用户的下一个任务都因 CPU 不足放不下\n",
		t.cpu, t.mem, capRes.cpu-t.cpu, capRes.mem-t.mem)

	// 均分：每用户固定 1/3 配额（4 核 / 8 GiB），配额内先到先得，不借用
	eq := newUsers()
	for _, u := range eq {
		quota := res{capRes.cpu / 3, capRes.mem / 3}
		for u.used.plus(u.task).fits(quota) {
			u.used = u.used.plus(u.task)
			u.n++
		}
	}

	// FIFO：B 先一次性提交 6 个任务（突刺），然后 A×6、C×6；放不下就跳过、不重排
	fifo := newUsers()
	var order []string
	for i := 0; i < 6; i++ {
		order = append(order, "B")
	}
	for i := 0; i < 6; i++ {
		order = append(order, "A")
	}
	for i := 0; i < 6; i++ {
		order = append(order, "C")
	}
	byName := map[string]*user{}
	for _, u := range fifo {
		byName[u.name] = u
	}
	var got res
	for _, name := range order {
		u := byName[name]
		if got.plus(u.task).fits(capRes) {
			got = got.plus(u.task)
			u.used = u.used.plus(u.task)
			u.n++
		}
	}

	fmt.Println("\n== 三策略对比（同一批 18 个任务）==")
	print := func(name string, us []*user) {
		t := usedAll(us)
		fmt.Printf("  %-4s 服务 %2d 个 | CPU %2g/12 (%3.0f%%) | 内存 %2g/24 (%3.0f%%) | 主导份额 %s/%s/%s\n",
			name, served(us), t.cpu, 100*t.cpu/capRes.cpu, t.mem, 100*t.mem/capRes.mem,
			f2(us[0].used.share()), f2(us[1].used.share()), f2(us[2].used.share()))
	}
	print("DRF", us)
	print("均分", eq)
	print("FIFO", fifo)
}

func dominant(cpuIsDominant bool) string {
	if cpuIsDominant {
		return "CPU"
	}
	return "内存"
}

func f2(v float64) string { return fmt.Sprintf("%.2f", v) }

                  真实输出（本机仿真）：

                  == 形状 → 主导份额 ==
  A 任务 (1核,4G)  cpu份额 0.083  内存份额 0.167  主导资源=内存  Δd=0.167
  B 任务 (3核,1G)  cpu份额 0.250  内存份额 0.042  主导资源=CPU  Δd=0.250
  C 任务 (2核,2G)  cpu份额 0.167  内存份额 0.083  主导资源=CPU  Δd=0.167

== DRF 分配序列（每步选主导份额最小且放得下者）==
  第 1 步 选中 A  d 0.000→0.167  该用户累计 (1核,4G)
  第 2 步 选中 B  d 0.000→0.250  该用户累计 (3核,1G)
  第 3 步 选中 C  d 0.000→0.167  该用户累计 (2核,2G)
  第 4 步 选中 A  d 0.167→0.333  该用户累计 (2核,8G)
  第 5 步 选中 C  d 0.167→0.333  该用户累计 (4核,4G)
  第 6 步 选中 B  d 0.250→0.500  该用户累计 (6核,2G)

== DRF 终局 ==
  A 任务 2 个  (2核,8G)  d=0.333
  B 任务 2 个  (6核,2G)  d=0.500
  C 任务 2 个  (4核,4G)  d=0.333
  合计 (12核,14G) 剩余 (0核,10G) → 三个用户的下一个任务都因 CPU 不足放不下

== 三策略对比（同一批 18 个任务）==
  DRF  服务  6 个 | CPU 12/12 (100%) | 内存 14/24 ( 58%) | 主导份额 0.33/0.50/0.33
  均分   服务  5 个 | CPU  9/12 ( 75%) | 内存 13/24 ( 54%) | 主导份额 0.33/0.25/0.33
  FIFO 服务  4 个 | CPU 12/12 (100%) | 内存  4/24 ( 17%) | 主导份额 0.00/1.00/0.00

                  完整代码在仓库 tools/examples/12.5-drf/，go run . 可复现这段输出。

                  序列里的前三步是平手（初始 d 都是 0），按固定顺序 A→B→C；第 4 步回到 A，因为此刻 A 的 d=0.167、C 的 d=0.167、B 的 d=0.250，A 与 C 平手时按用户顺序取 A；第 6 步轮到 B，因为 A、C 都已到 0.333，只剩 B 的 0.250 最小。手算核对第 4 步：A 再用一个 (1,4) 后累计 (2,8)，d = max(2/12, 8/24) = 0.333，与输出一致。

                  六步之后 CPU 归零、内存还剩 10 GiB，但三个用户的下一个任务都至少要 1 核 CPU。剩余资源放不下任何一个任务，调度就停在这里——闲置的 10 GiB 内存不是被谁浪费了，是被 CPU 这个瓶颈卡住的碎片。终局份额是 0.333 / 0.500 / 0.333，B 最高：不是 DRF 偏心，而是任务粒度决定的，B 每个任务就占 0.25，两次就到 0.5；A、C 各两个任务后下一个都放不下。DRF 保证的性质是“在放得下的前提下，没有谁能让另一个人的份额更低”，它不保证份额数字相等。

                  同一批 18 个任务（A/B/C 各 6 个，B 先一次性提交 6 个造成突刺）下，三种策略的差别：

                  策略 服务任务数 已用 CPU 已用内存 主导份额 A/B/C 最大/最小份额
                  DRF（主导份额最小者优先） 6 12/12 (100%) 14/24 (58%) 0.33 / 0.50 / 0.33 1.50
                  均分（固定 1/3 配额、不借用） 5 9/12 (75%) 13/24 (54%) 0.33 / 0.25 / 0.33 1.33
                  FIFO（按到达顺序、整块满足） 4 12/12 (100%) 4/24 (17%) 0.00 / 1.00 / 0.00 ∞

                  三个读法：

                  1. 服务数：DRF 6 > 均分 5 > FIFO 4。FIFO 只服务 4 个，是因为 B 的突发先占满 CPU，后面 A、C 的 1 核任务也排不进来；同时 20 GiB 内存全程闲着——先到先得的代价是“资源形状被第一个用户锁死”。
                  2. 公平性：均分的份额比 1.33 反而比 DRF 的 1.50 更齐。因为均分把每个用户钉在 1/3 配额上，谁也别想多拿；代价是 3 核 CPU（25%）锁在配额里，A、C 想借也借不到。“更公平”和“更死板”是同一枚硬币：均分放弃借用，就放弃了利用率。
                  3. FIFO 的问题不是“排队”，而是“整块满足 + 不重排”：它的份额比是无穷大（两人为零），因为调度器从头到尾没问过“谁该拿”，只问过“队列里下一个是谁”。

                  把这套判断搬到线上，盯三个数就够：服务任务数（吞吐）、主导份额最大/最小比（公平，经验上超过 2 且持续 10 分钟以上要查）、空闲资源形状（若内存空闲超过 30% 而 CPU 打满，说明任务形状与池子不匹配，加内存没用，要拆小任务或分池）。算例里 10 GiB 空闲内存（42%）就是典型信号：这不是“再加内存”能解的问题。

                  把这套判断搬到线上时，算法一个字都不用改，换的是三个输入：cap 换成节点池的可分配总量（不是物理容量，见 12.1 资源调度的目标与抽象）；“用户”换成 namespace、队列或租户；任务形状换成单 Pod 的 requests。真正要新增的是“多久重算一次”和“放不下时等还是抢”：YARN 的 Fair Scheduler 与 Mesos 的 DRF 分配器用的就是这个模型；K8s 默认调度器按优先级 + 过滤/打分选节点，不直接实现 DRF，而 ResourceQuota 更像上表里的“固定配额”——只封顶、不排序、也不允许借用。所以“公平”在 K8s 上要靠配额（硬边界）+ 优先级（12.6）+ 调度器插件（如按队列加权的插件）拼出来，而不是一个开关。

                  算例的边界也要说清：它假设 backlog 无限、任务形状固定、无抢占、无亲和/污点约束。真实集群里用户会撤任务、任务有优先级、资源还有 GPU 与本地盘，而且“放不下”常常是约束而不是容量。DRF 只回答“份额怎么算、下一个给谁”，不回答“资源不够时该杀谁”——后者归抢占与排队（12.6 抢占、优先级与排队）。当集群里出现“份额看着公平、但用户还是抱怨”时，先查任务粒度：把大任务拆小，份额分辨率和可服务任务数会一起上去，这个算例里 B 的 0.25 粒度就是最直接的反例。

                  反模式

                  反模式 后果
                  用 “平均分 CPU” 做多租户公平 内存型任务把节点 OOM，CPU 却空闲
                  只按提交顺序排队 大任务堵住队列，小任务延迟爆炸
                  不设 “最大份额上限” 单一用户可占满全池，公平失效
                  配额按 “团队” 而不是 “队列” 团队内部互相抢，外部看不到
                  忽略资源形状（只算总量） 碎片化：总量够但放不下
                  公平性不做监控 无法证明 “谁被饿死”

                  三条铁律

                  🔑 公平要按 “主导份额” 比，不按单维资源比。

                  🔑 公平与利用率是对偶的：越公平，装箱越碎、利用率可能越低。

                  🔑 配额必须有上限与借用规则，否则 “公平” 只是口号。

                  自检

                  Q1 DRF 的 “主导份额” 为什么能解决 “形状不同不可比” 的问题？

                  ▸ 答案：因为它把 “绝对用量” 换成 “占全局容量的比例”：A 用 1 核 4G 在 16/16 的池子里主导份额是 0.25（内存），B 用 4 核 1G 也是 0.25（CPU）。两者虽然用量不同，但 “主导份额” 相等，就可以比较 “谁更该拿下一份资源”。

                  Q2 实测中 FIFO 只服务了 10 个任务，为什么这么少？

                  ▸ 答案：因为 FIFO 按 “任务到达顺序 + 整块满足” 分配：A（1核4G）先来，连续占满内存维度后，B（4核1G）虽然 CPU 还够，但整块申请被 “内存不足” 挡住，于是后续任务全部排队。FIFO 的浪费来自 “不考虑形状”，不是容量不够。

                  Q3 DRF 会不会让 “小任务” 一直占优势？

                  ▸ 答案：不会，因为份额是累计的：小任务拿得越多，主导份额越高，后续就越排在后面。DRF 保证的是 “长期份额趋近相等”，不是 “小任务优先”。但如果小任务数量极多，份额增长慢，仍可能出现 “大任务等待较久”——这需要 “最大等待时间/老化” 机制补充。

                  Q4 公平调度与 “优先级” 能共存吗？

                  ▸ 答案：能，常见做法是 “先按优先级分层，层内按 DRF 公平”：高优先级队列先拿资源，层内再公平分配。代价是低优先级可能长期拿不到——所以要有 “配额保证/最小份额” 兜底，否则公平只是高优先级内部的公平。

                  Q5 为什么 “公平” 有时会降低利用率？

                  ▸ 答案：因为公平要求 “每个用户都能拿到一份”，即使某些用户的任务形状会留下碎片；而 “贪心装箱” 可以把碎片填满。所以公平与利用率是一对权衡：单租户追求利用率，多租户追求公平 + 可预测。

                  Q6 多资源里 “归一化” 的基准该用总容量还是可用容量？

                  ▸ 答案：用总容量（集群可分配总量）更稳定：可用容量随其他任务波动，会让 “份额” 抖动，导致调度决策反复。代价是 “集群扩容后份额需要重算”，这通常由配置中心/调度器周期刷新解决。

                  Q7 怎么验证 “公平真的生效了”？

                  ▸ 答案：看三条曲线：每个用户的主导份额（应收敛）、最大与最小份额的比值（应接近 1）、每用户等待时间分布（不应有长期饥饿）。只看 “平均利用率” 看不出公平问题——平均可以很漂亮，而某个用户一直排不上。
